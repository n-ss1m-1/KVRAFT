cmake_minimum_required(VERSION 3.16)
project(cache_accept LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
foreach(n internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest block_cache)
 list(APPEND SOURCES ${ENGINE_ROOT}/lib/${n}.cc)
endforeach()
add_library(engine STATIC ${SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(cache-bench bench.cc)
target_link_libraries(cache-bench PRIVATE engine)
add_executable(cache-mechanism bench.cc)
target_compile_definitions(cache-mechanism PRIVATE MECHANISM=1)
target_link_libraries(cache-mechanism PRIVATE engine)
target_link_options(cache-mechanism PRIVATE "-Wl,--wrap=_ZN10BlockCache3GetERK13BlockCacheKey" "-Wl,--wrap=_ZN10BlockCache6InsertERK13BlockCacheKeySt10shared_ptrIK15DataBlockReaderEm" "-Wl,--wrap=pread")

target_link_options(cache-bench PRIVATE "-Wl,--wrap=_ZN10BlockCacheC1Em")
target_link_options(cache-mechanism PRIVATE "-Wl,--wrap=_ZN10BlockCacheC1Em")

cmake_minimum_required(VERSION 3.16)
project(compaction_diagnosis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(LSM_DIAGNOSTICS "existing diagnostics" OFF)
if(LSM_DIAGNOSTICS)
 message(FATAL_ERROR "Use independent probe, existing fine timings disabled")
endif()
foreach(kind plain probe)
 set(SOURCES)
 if(kind STREQUAL plain)
  set(E ${RESULT_ROOT}/original)
 else()
  set(E ${RESULT_ROOT}/instrumented)
 endif()
 foreach(n internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest block_cache write_batch compaction)
  list(APPEND SOURCES ${E}/lib/${n}.cc)
 endforeach()
 add_library(engine_${kind} STATIC ${SOURCES})
 target_include_directories(engine_${kind} PUBLIC ${E}/include PRIVATE ${E}/include/lsm ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_options(engine_${kind} PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/dd.h)
 add_executable(bench_${kind} bench.cc)
 target_link_libraries(bench_${kind} PRIVATE engine_${kind} Threads::Threads)
 if(kind STREQUAL probe)
  target_compile_definitions(engine_${kind} PUBLIC PROBE=1)
 endif()
endforeach()

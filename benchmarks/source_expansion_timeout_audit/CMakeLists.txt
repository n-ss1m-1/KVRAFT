cmake_minimum_required(VERSION 3.16)
project(timeout_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(ENGINE_ROOT "" CACHE PATH "Historical source-expansion observer snapshot")
if(NOT EXISTS "${ENGINE_ROOT}/lib/db.cc")
    message(FATAL_ERROR "Set ENGINE_ROOT to the original source-expansion observer snapshot")
endif()
file(GLOB sources "${ENGINE_ROOT}/lib/*.cc")
add_executable(audit audit.cc ${sources})
target_include_directories(audit PRIVATE "${ENGINE_ROOT}/include"
    "${ENGINE_ROOT}/include/lsm" "${CMAKE_CURRENT_SOURCE_DIR}/../source_expansion_resume")
target_compile_definitions(audit PRIVATE OBSERVE=1)
target_link_libraries(audit PRIVATE Threads::Threads)

cmake_minimum_required(VERSION 3.16)
project(crc_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
foreach(name internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest)
 list(APPEND SOURCES ${ENGINE_ROOT}/lib/${name}.cc)
endforeach()
add_library(engine STATIC ${SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(acceptance acceptance.cc)
target_link_libraries(acceptance PRIVATE engine)
add_executable(crc-micro crc_micro.cc)
target_link_libraries(crc-micro PRIVATE engine)

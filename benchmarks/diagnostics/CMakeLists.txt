cmake_minimum_required(VERSION 3.16)
project(lsm_diagnostics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LSM_DIAGNOSTICS "Enable cost counters" OFF)
find_package(Threads REQUIRED)
set(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../src/lsm" CACHE PATH "Engine sources")
foreach(name internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest)
 list(APPEND SOURCES ${ENGINE_ROOT}/lib/${name}.cc)
endforeach()
add_library(engine STATIC ${SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
if(LSM_DIAGNOSTICS)
 target_compile_definitions(engine PUBLIC LSM_DIAGNOSTICS=1)
endif()
add_executable(diagnose diagnose.cc)
target_link_libraries(diagnose PRIVATE engine)
target_compile_options(diagnose PRIVATE -Wall -Wextra -Wpedantic)
add_executable(stats-check stats_check.cc)
target_link_libraries(stats-check PRIVATE engine)

add_executable(followup-write followup_write.cc)
target_link_libraries(followup-write PRIVATE engine)
add_executable(crc-bench crc_bench.cc)
target_link_libraries(crc-bench PRIVATE engine)

cmake_minimum_required(VERSION 3.16)
project(dual_background LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(LSM_DIAGNOSTICS "diagnostics" OFF)
if(LSM_DIAGNOSTICS)
 message(FATAL_ERROR "diagnostics must be OFF")
endif()
foreach(kind serial current-serial engine-snapshot serial-probe engine-snapshot-probe)
 set(SOURCES)
 foreach(n internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest block_cache write_batch compaction)
  list(APPEND SOURCES ${RESULT_ROOT}/${kind}/lib/${n}.cc)
 endforeach()
 add_library(e_${kind} STATIC ${SOURCES})
 target_include_directories(e_${kind} PUBLIC ${RESULT_ROOT}/${kind}/include PRIVATE ${RESULT_ROOT}/${kind}/include/lsm ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_options(e_${kind} PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/dd.h)
 if(kind MATCHES "probe")
  target_compile_definitions(e_${kind} PUBLIC PROBE=1)
 endif()
 add_executable(bench_${kind} bench.cc)
 target_link_libraries(bench_${kind} PRIVATE e_${kind} Threads::Threads)
endforeach()

cmake_minimum_required(VERSION 3.16)
project(lsm_compare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(NOT ENGINE_ROOT)
 message(FATAL_ERROR "Set ENGINE_ROOT to isolated snapshot src/lsm")
endif()
set(ENGINE_FILES internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest)
foreach(name IN LISTS ENGINE_FILES)
 list(APPEND ENGINE_SOURCES ${ENGINE_ROOT}/lib/${name}.cc)
endforeach()
add_library(engine STATIC ${ENGINE_SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(compare_bench compare_bench.cc)
target_link_libraries(compare_bench PRIVATE engine)
target_compile_options(compare_bench PRIVATE -Wall -Wextra -Wpedantic)

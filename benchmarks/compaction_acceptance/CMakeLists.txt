cmake_minimum_required(VERSION 3.16)
project(compaction_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(LSM_DIAGNOSTICS "diagnostics" OFF)
if(LSM_DIAGNOSTICS)
 message(FATAL_ERROR "diagnostics must be OFF")
endif()
find_package(Threads REQUIRED)
foreach(n internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest block_cache write_batch compaction)
 list(APPEND SOURCES ${ENGINE_ROOT}/lib/${n}.cc)
endforeach()
add_library(engine STATIC ${SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(compaction-bench bench.cc)
target_link_libraries(compaction-bench PRIVATE engine)

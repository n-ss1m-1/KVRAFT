cmake_minimum_required(VERSION 3.16)
project(batch_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
foreach(n internal_key skip_list mem_table coding wal bloom_filter data_block sstable db manifest block_cache write_batch)
 list(APPEND SOURCES ${ENGINE_ROOT}/lib/${n}.cc)
endforeach()
add_library(engine STATIC ${SOURCES})
target_include_directories(engine PUBLIC ${ENGINE_ROOT}/include PRIVATE ${ENGINE_ROOT}/include/lsm)
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(batch-bench bench.cc)
target_link_libraries(batch-bench PRIVATE engine)
add_executable(batch-mechanism bench.cc)
target_compile_definitions(batch-mechanism PRIVATE MECHANISM=1)
target_link_libraries(batch-mechanism PRIVATE engine)
target_link_options(batch-mechanism PRIVATE "-Wl,--wrap=fsync" "-Wl,--wrap=write")
